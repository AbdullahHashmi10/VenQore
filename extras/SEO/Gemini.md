    '/compare/toast-pos-alternative',
    '/compare/quickbooks-pos-migration',
    '/integrations/woocommerce-pos-sync',
    '/tools/retail-gross-margin-calculator',
    '/tools/inventory-turnover-calculator',
    '/tools/free-online-barcode-generator',
  ];

  const routes = staticPages.map((route) => {
    let priority = 0.8;
    if (route === '') priority = 1.0;

      priority,
    };
  });

  return routes;
}
